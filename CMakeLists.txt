cmake_minimum_required(VERSION 3.24)
project(CivCTP2 VERSION 1.2 LANGUAGES C CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use a separate build directory: cmake -S . -B build/native")
endif()
include(GNUInstallDirs)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(CTP2_BUILD_GAME "Build the SDL2 game and map plugins" ON)
option(CTP2_SANITIZERS "Instrument project targets with ASan and UBSan" OFF)
option(CTP2_LOGGING "Enable game logging" OFF)
option(CTP2_MOVIES "Enable optional FFmpeg video support" OFF)
set(CTP2_HOST_DBGEN "" CACHE FILEPATH "Host-native ctpdb executable for cross builds")
set(USE_LOGGING ${CTP2_LOGGING})
set(USE_SDL_FFMPEG ${CTP2_MOVIES})
configure_file(cmake/ctp2_build_config.h.in generated/config/ctp2_build_config.h @ONLY)
add_library(ctp2_options INTERFACE)
target_include_directories(ctp2_options INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/generated/config" "${CMAKE_CURRENT_SOURCE_DIR}/ctp2_code/os/include")
target_compile_definitions(ctp2_options INTERFACE CTP2_CMAKE_BUILD
  $<$<CONFIG:Debug>:_DEBUG;_PLAYTEST;_DEBUGTOOLS;_DEBUG_MEMORY>
  $<$<NOT:$<CONFIG:Debug>>:_BFR_>)
if(MSVC)
  target_compile_options(ctp2_options INTERFACE /W3)
  target_compile_definitions(ctp2_options INTERFACE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
else()
  target_compile_options(ctp2_options INTERFACE -fms-extensions -Wall -Wextra -Wno-deprecated-declarations)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(ctp2_options INTERFACE "$<$<COMPILE_LANGUAGE:CXX>:-Werror=non-pod-varargs>")
  endif()
endif()
if(CTP2_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" OR WIN32)
    message(FATAL_ERROR "This diagnostic preset requires Clang/GCC ASan+UBSan on Unix")
  endif()
  target_compile_options(ctp2_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(ctp2_options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=all)
endif()
include(cmake/Generate.cmake)
if(CTP2_BUILD_GAME)
  include(cmake/Dependencies.cmake)
  include(cmake/Game.cmake)
endif()
if(BUILD_TESTING)
  foreach(test IN ITEMS native-abi slic-bytecode tile-binary)
    add_executable(${test}-test tests/${test}.cpp)
    target_link_libraries(${test}-test PRIVATE ctp2_options)
    # These tests use assertions in every configuration.
    target_compile_options(${test}-test PRIVATE $<IF:$<CXX_COMPILER_ID:MSVC>,/UNDEBUG,-UNDEBUG>)
    add_test(NAME ${test} COMMAND ${test}-test)
  endforeach()
endif()
